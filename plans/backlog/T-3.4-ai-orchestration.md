# T-3.4 — AI Orchestration

**Phase:** AI Pipeline
**Layer:** Backend
**Status:** Backlog
**Estimated Effort:** 2–3 hours
**Dependencies:** T-3.1, T-3.2, T-3.3

---

## Objective

Implement the AI orchestrator that manages the tool-calling loop: builds the prompt, calls the LLM, validates and executes tools, feeds results back, and enforces iteration limits.

---

## Acceptance Criteria

- [ ] `AIOrchestratorService` that orchestrates a full conversation turn
- [ ] System prompt includes current Colombia date/time and available tools
- [ ] Conversation history loaded from MongoDB and included in prompt
- [ ] Tool-calling loop: LLM → validate → execute → feed back → repeat
- [ ] Maximum 5 iterations per turn; if reached, force escalation
- [ ] Each tool call and result traced and saved to `ai_traces` collection
- [ ] Handles LLM errors gracefully (timeout → escalation)
- [ ] Returns final response content and conversation status
- [ ] `AIModule` integrating orchestrator, LLM service, embedding service, tool validator

### 2. AI Orchestrator Service

Create `src/application/llm/ai-orchestrator.service.ts`:
```typescript
@Injectable()
export class AIOrchestratorService {
  constructor(
    @Inject(LLM_SERVICE) private llmService: LLMService,
    @Inject(KNOWLEDGE_REPOSITORY) private knowledgeRepo: KnowledgeRepository,
    @Inject(SLOT_REPOSITORY) private slotRepo: SlotRepository,
    @Inject(APPOINTMENT_REPOSITORY) private appointmentRepo: AppointmentRepository,
    @Inject(AI_TRACE_REPOSITORY) private aiTraceRepo: AITraceRepository,
    private toolValidator: ToolValidator,
    private promptBuilder: PromptBuilder,
  ) {}

  async processTurn(conversationId: string, clinicId: string, userMessage: string): Promise<{ response: string; status: ConversationStatus }> {
    const MAX_ITERATIONS = 5;
    const history = await this.messageRepo.findByConversationId(conversationId);
    const systemPrompt = this.promptBuilder.buildSystemPrompt(clinicName);
    const messages = this.promptBuilder.buildMessages(history, userMessage);
    const trace = { toolsCalled: [], turnIndex: history.length };

    for (let i = 0; i < MAX_ITERATIONS; i++) {
      const result = await this.llmService.chat({ messages, tools: TOOL_DEFINITIONS });

      if (result.toolCalls.length === 0) {
        // No tool calls → return content
        await this.saveTrace(conversationId, clinicId, trace, result, 'resolved_by_ai');
        return { response: result.content, status: 'resolved_by_ai' };
      }

      // Process each tool call
      for (const toolCall of result.toolCalls) {
        const validation = this.toolValidator.validate(toolCall.name, toolCall.arguments);
        if (!validation.valid) {
          // Return error to LLM as tool result
          messages.push({ role: 'assistant', tool_calls: [toolCall] });
          messages.push({ role: 'tool', tool_call_id: toolCall.id, content: validation.error });
          trace.toolsCalled.push({ name: toolCall.name, arguments: toolCall.arguments, result: validation.error, success: false });
          continue;
        }

        const toolResult = await this.executeTool(toolCall.name, validation.parsed, clinicId);
        messages.push({ role: 'assistant', tool_calls: [toolCall] });
        messages.push({ role: 'tool', tool_call_id: toolCall.id, content: JSON.stringify(toolResult) });
        trace.toolsCalled.push({ name: toolCall.name, arguments: toolCall.arguments, result: toolResult, success: true });

        // Check if appointment was booked or escalated
        if (toolCall.name === 'agendar_cita') {
          await this.saveTrace(conversationId, clinicId, trace, result, 'appointment_booked');
          return { response: 'Cita agendada exitosamente.', status: 'appointment_booked' };
        }
        if (toolCall.name === 'escalar_a_humano') {
          await this.saveTrace(conversationId, clinicId, trace, result, 'escalated');
          return { response: 'Te estoy transfiriendo a un agente humano.', status: 'escalated' };
        }
      }
    }

    // Max iterations reached → force escalation
    await this.executeTool('escalar_a_humano', { motivo: 'Límite de iteraciones alcanzado' }, clinicId);
    return { response: 'No pude resolver tu consulta. Te transfiero a un agente.', status: 'escalated' };
  }

  private async executeTool(name: string, args: any, clinicId: string): Promise<any> {
    switch (name) {
      case 'buscar_conocimiento': return this.knowledgeRepo.search(clinicId, args.pregunta);
      case 'consultar_disponibilidad': return this.slotRepo.findAvailable(clinicId, args.especialidad, new Date(args.fecha));
      case 'agendar_cita': return this.bookAppointment(args, clinicId);
      case 'escalar_a_humano': return { escalated: true, reason: args.motivo };
    }
  }
}
```
