# T-9.1 — Backend Logging System

**Phase:** Logging
**Layer:** Backend
**Status:** Completed
**Estimated Effort:** 2-3 hours
**Dependencies:** T-5.2
**Completed:** 2026-10-03

---

## Objective

Implement a structured logging system for the NestJS backend using Winston. The logger must be abstracted behind a domain-level interface (following the existing DI token pattern) so the logging provider can be swapped. HTTP request/response logging must capture method, path, status, duration, and sanitize sensitive body fields.

---

## Acceptance Criteria

- [x] `Logger` interface in domain layer with `debug`, `info`, `warn`, `error` methods
- [x] `LOGGER` DI token for injecting the logger (consistent with existing token pattern in `@domain/repositories` and `@domain/services`)
- [x] `WinstonLoggerService` implementing `Logger` and NestJS `LoggerService`
- [x] Winston configuration: colorized console in development, JSON in production
- [x] Daily rotate file transport for production (`logs/error-*.log`, `logs/combined-*.log`)
- [x] `LoggingInterceptor` logging every HTTP request/response with method, path, status, duration
- [x] Request body sanitized (password, token, secret, apiKey fields redacted)
- [x] `formatLog` utility for consistent log structure
- [x] `LoggerModule` (global) exporting `LOGGER` token
- [x] Error logs include stack trace
- [x] Log level configurable via `NODE_ENV` (`debug` in dev, `info` in prod)
- [ ] All existing controllers and services use `@Inject(LOGGER)` for logging

## Implementation Steps

### 1. Logger Interface (Domain)

Create `apps/api/src/domain/services/logger.interface.ts`:
```typescript
export const LOGGER = Symbol('LOGGER');

export type LogMetadata = Record<string, unknown>;

export interface Logger {
  debug(message: string, metadata?: LogMetadata): void;
  info(message: string, metadata?: LogMetadata): void;
  warn(message: string, metadata?: LogMetadata): void;
  error(message: string, metadata?: LogMetadata): void;
}
```

### 2. Log Formatter

Create `apps/api/src/infrastructure/logging/logger.formatter.ts`:
```typescript
export interface FormattedLog {
  message: string;
  metadata: Record<string, unknown>;
}

export function formatLog(message: string, metadata?: Record<string, unknown>): FormattedLog {
  return { message, metadata: metadata ?? {} };
}

export function stringifyLogValue(value: unknown): string {
  if (typeof value === 'string') return value;
  if (value === undefined || value === null) return '';
  try { return JSON.stringify(value); } catch { return '[Unserializable value]'; }
}
```

### 3. Winston Configuration

Install dependencies:
```bash
pnpm --filter api add winston winston-daily-rotate-file
```

Create `apps/api/src/infrastructure/logging/winston/winston.config.ts`:
- Console transport with colorized format in dev
- JSON format for file transports in prod
- Daily rotate file for errors (`logs/error-%DATE%.log`)
- Daily rotate file for combined (`logs/combined-%DATE%.log`)
- Max size 20MB, retention 14 days

### 4. Winston Logger Service

Create `apps/api/src/infrastructure/logging/winston/winston-logger.service.ts`:
```typescript
@Injectable()
export class WinstonLoggerService implements LoggerService, Logger {
  private readonly logger: winston.Logger;

  constructor(private readonly configService: ConfigService) {
    const isProduction = configService.get('NODE_ENV') === 'production';
    this.logger = winston.createLogger(createWinstonConfig(isProduction));
  }

  debug(message: string, metadata?: LogMetadata): void { ... }
  info(message: string, metadata?: LogMetadata): void { ... }
  warn(message: string, metadata?: LogMetadata): void { ... }
  error(message: string, metadata?: LogMetadata): void { ... }
  log(message: string): void { this.info(message); }
}
```

### 5. Logging Interceptor

Create `apps/api/src/presentation/interceptors/logging.interceptor.ts`:
```typescript
@Injectable()
export class LoggingInterceptor implements NestInterceptor {
  constructor(@Inject(LOGGER) private readonly logger: Logger) {}

  intercept(context: ExecutionContext, next: CallHandler): Observable<any> {
    const request = context.switchToHttp().getRequest<Request>();
    const { method, url, body } = request;
    const start = Date.now();

    this.logger.info('Incoming request', {
      context: 'HTTP', method, path: url,
      body: this.sanitizeBody(body),
    });

    return next.handle().pipe(
      tap({
        next: () => {
          const response = context.switchToHttp().getResponse();
          this.logger.info('Response sent', {
            context: 'HTTP', method, path: url,
            statusCode: response.statusCode,
            duration: Date.now() - start,
          });
        },
        error: (error) => {
          const response = context.switchToHttp().getResponse();
          this.logger.error('Request failed', {
            context: 'HTTP', method, path: url,
            statusCode: response.statusCode || 500,
            duration: Date.now() - start,
            error: error.message, stack: error.stack,
          });
        },
      }),
    );
  }

  private sanitizeBody(body: unknown): Record<string, unknown> {
    // Redact password, token, secret, apiKey, key fields
  }
}
```

### 6. Logger Module (Global)

Create `apps/api/src/infrastructure/logging/logger.module.ts`:
```typescript
@Global()
@Module({
  providers: [
    { provide: LOGGER, useClass: WinstonLoggerService },
    WinstonLoggerService,
  ],
  exports: [LOGGER, WinstonLoggerService],
})
export class LoggerModule {}
```

### 7. Register in App Module

Import `LoggerModule` in `apps/api/src/app.module.ts`. Register `LoggingInterceptor` in `apps/api/src/main.ts`:
```typescript
app.useGlobalInterceptors(new LoggingInterceptor(app.get(LOGGER)));
```
